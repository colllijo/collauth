105{
106 digits.clear();
107
108 if constexpr (std::is_signed_v<T>)

111 uint64_t value = static_cast<uint64_t>(static_cast<std::make_unsigned_t<T>>(negative ? -(number + 1) + 1 : number));

114 {
115 digits.push_back(static_cast<uint32_t>(value & MASK32));
116 value >>= 32;
117 } while (value != 0);
